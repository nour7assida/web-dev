START

Read character

length ← 0
words ← 1
vowels ← 0

WHILE character ≠ "."

    length ← length + 1

    IF character = " " THEN
        words ← words + 1
    END IF

    IF character = "a" OR character = "e" OR character = "i" OR character = "o" OR character = "u" THEN
        vowels ← vowels + 1
    END IF

    Read character

END WHILE

length ← length + 1

Display length
Display words
Display vowels

END
